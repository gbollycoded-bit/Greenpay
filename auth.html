async function signup() { clearMessage(); const rpcCode = document.getElementById("rpcCode").value.trim(); const fullName = document.getElementById("fullName").value.trim(); const email = document.getElementById("signupEmail").value.trim(); const password = document.getElementById("signupPassword").value; const button = document.getElementById("signupButton"); if (!rpcCode || !fullName || !email || !password) { showMessage( "Please complete all fields.", "error" ); return; } if (password.length < 6) { showMessage( "Password must contain at least 6 characters.", "error" ); return; } button.disabled = true; button.textContent = "Checking RPC code..."; try { // STEP 1: // Check the RPC code BEFORE creating an account. const { data: validCode, error: checkError } = await client.rpc( "check_rpc_code", { rpc_code: rpcCode } ); if (checkError) { throw checkError; } if (validCode !== true) { showMessage( "Invalid or already-used RPC activation code.", "error" ); button.disabled = false; button.textContent = "Activate & Create Account"; return; } // STEP 2: // RPC code is valid, so create the account. button.textContent = "Creating account..."; const { data, error } = await client.auth.signUp({ email: email, password: password, options: { data: { full_name: fullName } } }); if (error) { throw error; } if (!data.user) { throw new Error( "Account could not be created." ); } // STEP 3: // Mark the RPC code as used. button.textContent = "Activating RPC..."; const { data: activated, error: activateError } = await client.rpc( "activate_rpc_code", { rpc_code: rpcCode } ); if (activateError) { throw activateError; } if (activated !== true) { showMessage( "The RPC code could not be activated. Please contact support.", "error" ); return; } // Save account information. await client.auth.updateUser({ data: { full_name: fullName, rpc_activated: true } }); showMessage( "Account created successfully!", "success" ); setTimeout(() => { window.location.href = "dashboard.html"; }, 1000); } catch (error) { console.error(error); showMessage( error.message || "Something went wrong. Please try again.", "error" ); } finally { button.disabled = false; button.textContent = "Activate & Create Account"; } }
